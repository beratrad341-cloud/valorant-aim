    <script>
        function checkAuth() {
            const savedUser = localStorage.getItem('valorant_user');
            if (savedUser) {
                document.getElementById('authModal').classList.add('hidden');
                document.getElementById('mainContent').classList.remove('bg-blur-lock');
            } else {
                document.getElementById('authModal').classList.remove('hidden');
                document.getElementById('mainContent').classList.add('bg-blur-lock');
            }
        }

        function handleAuth(e) {
            e.preventDefault();
            const username = document.getElementById('usernameInput').value.trim();
            if (username) {
                localStorage.setItem('valorant_user', username);
                checkAuth();
            }
        }

        window.addEventListener('DOMContentLoaded', checkAuth);

        let score = 0;
        let timeLeft = 30;
        let shotsFired = 0;
        let shotsHit = 0;
        let gameInterval = null;
        let targetTimeout = null;

        function startGame() {
            const arena = document.getElementById('gameArena');
            arena.innerHTML = '';
            score = 0;
            timeLeft = 30;
            shotsFired = 0;
            shotsHit = 0;
            updateHUD();

            arena.onclick = (e) => {
                if (e.target.id === 'gameArena') {
                    shotsFired++;
                    updateAccuracy();
                }
            };

            clearInterval(gameInterval);
            gameInterval = setInterval(() => {
                timeLeft--;
                updateHUD();
                if (timeLeft <= 0) {
                    endGame();
                }
            }, 1000);

            spawnBotTarget();
        }

        function spawnBotTarget() {
            if (timeLeft <= 0) return;
            const arena = document.getElementById('gameArena');
            arena.innerHTML = '';

            const diff = document.getElementById('diffSelect').value;
            let size = 55;
            if (diff === 'easy') size = 70;
            if (diff === 'medium') size = 55;
            if (diff === 'hard') size = 40;

            const bot = document.createElement('div');
            bot.style.width = `${size}px`;
            bot.style.height = `${size + 15}px`;
            bot.style.position = 'absolute';
            
            const maxX = arena.clientWidth - size - 20;
            const maxY = arena.clientHeight - size - 30;
            const randomX = Math.floor(Math.random() * maxX);
            const randomY = Math.floor(Math.random() * maxY);

            bot.style.left = `${randomX}px`;
            bot.style.top = `${randomY}px`;
            bot.className = 'bg-[#ff4655] rounded-t-full border-2 border-white cursor-pointer shadow-lg shadow-[#ff4655]/50 flex flex-col items-center justify-start pt-1 transition-transform transform hover:scale-105';
            bot.innerHTML = `
                <div class="w-3 h-3 bg-yellow-300 rounded-full mb-1 border border-black" title="Kafa Vuruşu"></div>
                <div class="text-[9px] font-bold text-white uppercase tracking-tighter">BOT</div>
            `;

            bot.onclick = (e) => {
                e.stopPropagation();
                shotsFired++;
                shotsHit++;
                
                let points = 1;
                if (diff === 'medium') points = 2;
                if (diff === 'hard') points = 3;
                
                score += points;
                updateAccuracy();
                updateHUD();
                spawnBotTarget();
            };

            arena.appendChild(bot);

            let speed = 1100;
            if (diff === 'medium') speed = 850;
            if (diff === 'hard') speed = 650;

            clearTimeout(targetTimeout);
            targetTimeout = setTimeout(() => {
                if (timeLeft > 0) {
                    shotsFired++;
                    updateAccuracy();
                    spawnBotTarget();
                }
            }, speed);
        }

        function updateHUD() {
            document.getElementById('scoreVal').innerText = score;
            document.getElementById('timeVal').innerText = timeLeft + 's';
        }

        function updateAccuracy() {
            let acc = shotsFired > 0 ? Math.round((shotsHit / shotsFired) * 100) : 100;
            document.getElementById('accVal').innerText = `%${acc}`;
        }

        function endGame() {
            clearInterval(gameInterval);
            clearTimeout(targetTimeout);
            const arena = document.getElementById('gameArena');
            let acc = shotsFired > 0 ? Math.round((shotsHit / shotsFired) * 100) : 100;
